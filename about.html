(function () { const button = document.getElementById('flipCoin'); const result = document.getElementById('coinResult'); const status = document.getElementById('coinStatus'); const coin = document.getElementById('coinVisual'); const flight = document.getElementById('coinFlight'); const hand = document.getElementById('coinHand'); if (!button || !result || !status || !coin || !flight || !hand) return; let running = false; let rafId = 0; let endAt = 0; let targetY = 0; let lastTimestamp = 0; const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches; function easeOutCubic(t) { return 1 - Math.pow(1 - t, 3); } function easeInOutCubic(t) { return t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2; } function resetCoin() { flight.style.transform = 'translate3d(0, 0, 0)'; coin.style.transform = 'rotateY(0deg) rotateZ(0deg)'; coin.classList.remove('coin-spark'); hand.classList.remove('hand-flicking'); } function finish(heads) { running = false; cancelAnimationFrame(rafId); const finalAngle = heads ? 3600 : 3780; // F: front, B: back coin.style.transform = `rotateY(${finalAngle}deg) rotateZ(0deg)`; flight.style.transform = 'translate3d(90px, 10px, 0)'; coin.classList.remove('coin-flying'); coin.classList.add('coin-settle', 'coin-spark'); result.textContent = heads ? '앞면 · F' : '뒷면 · B'; result.className = 'result-value coin-result-show'; status.textContent = heads ? 'F 면이 정면으로 멈췄습니다.' : 'B 면이 정면으로 멈췄습니다.'; button.disabled = false; button.textContent = '🪙 다시 던지기'; hand.classList.remove('hand-flicking'); } function animate(now) { if (!running) return; const p = Math.min((now - (endAt - 1850)) / 1850, 1); const spinProgress = easeInOutCubic(p); const travelProgress = easeOutCubic(p); // Continuous, frame-by-frame motion: no CSS keyframe stops or pauses. const angle = targetY * spinProgress; const arc = Math.sin(Math.PI * travelProgress); const x = 90 * travelProgress; const y = -125 * arc + 10; const z = 1 + 0.15 * Math.sin(Math.PI * travelProgress); const tilt = 9 * Math.sin(Math.PI * travelProgress * 1.25); coin.style.transform = `rotateY(${angle}deg) rotateZ(${tilt}deg) scale(${z})`; flight.style.transform = `translate3d(${x}px, ${y}px, 0)`; if (p < 1) { rafId = requestAnimationFrame(animate); } else { finish(targetY % 360 === 0); } } function toss() { if (running) return; running = true; const heads = Math.random() < 0.5; targetY = heads ? 3600 : 3780; endAt = performance.now() + 1850; lastTimestamp = 0; button.disabled = true; button.textContent = '🪙 던지는 중…'; result.textContent = '...'; result.className = 'result-value coin-result-wait'; status.textContent = '손가락을 튕겨 동전이 공중으로 올라가는 중입니다.'; coin.classList.remove('coin-settle', 'coin-spark'); coin.classList.add('coin-flying'); hand.classList.remove('hand-flicking'); void hand.offsetWidth; hand.classList.add('hand-flicking'); if (prefersReducedMotion) { window.setTimeout(() => finish(heads), 140); return; } cancelAnimationFrame(rafId); rafId = requestAnimationFrame((timestamp) => { lastTimestamp = timestamp; animate(timestamp); }); } resetCoin(); button.addEventListener('click', toss); })();
